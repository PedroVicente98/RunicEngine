cmake_minimum_required(VERSION 3.24)
project(FixtureHeaderOnly LANGUAGES CXX)
include("${CMAKE_CURRENT_LIST_DIR}/../FixturePrologue.cmake")
runic_catalog_module(Alpha GROUP Foundation HEADLESS yes)
runic_catalog_module(Beta GROUP Foundation ALLOWS Fixture::Alpha HEADLESS yes)
add_subdirectory(Alpha)
add_subdirectory(Beta)
get_target_property(alpha_real Fixture::Alpha ALIASED_TARGET)
get_target_property(alpha_type ${alpha_real} TYPE)
if(NOT alpha_type STREQUAL "INTERFACE_LIBRARY")
    message(FATAL_ERROR "fixture: a module with only Public/ must be an INTERFACE library, got ${alpha_type}")
endif()
