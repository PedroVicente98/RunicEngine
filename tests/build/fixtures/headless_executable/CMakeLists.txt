cmake_minimum_required(VERSION 3.24)
project(FixtureHeadlessExecutable LANGUAGES CXX)
include("${CMAKE_CURRENT_LIST_DIR}/../FixturePrologue.cmake")
option(FIXTURE_CLEAN "Leave out the presentation edge" OFF)
add_library(fake_glfw INTERFACE)
add_library(fake_bundle INTERFACE)
target_link_libraries(fake_bundle INTERFACE fake_glfw)
runic_mark_presentation(fake_glfw)
add_executable(FixtureServer main.cpp)
if(NOT FIXTURE_CLEAN)
    target_link_libraries(FixtureServer PRIVATE fake_bundle)
endif()
runic_check_headless(FixtureServer)
